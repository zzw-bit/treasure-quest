async function findTreasure() {
  const clue = await TreasureMap.getInitialClue();
  Bus.emit(EVT.ITEM_FOUND, clue);          // 通知界面

  let script;
  try {                                   // 失败就降级
    script = await TreasureMap.decodeAncientScript(clue);
  } catch (e) {
    Bus.emit(EVT.LOG, { text: e.message });
    script = await TreasureMap.askOldScholar(gold);
  }

  for (let i = 1; i <= 3; i++) {             // 重试 3 次
    try { key = await TreasureMap.findTempleKey(script); break; }
    catch (e) { if (i === 3) key = await TreasureMap.staticBackupKey(); }
  }
  return await TreasureMap.openTreasureBox();
}

function findTreasureWithPromises() {
  return TreasureMap.getInitialClue()
    .then((c) => TreasureMap.decodeAncientScript(c))
    .catch((e) => TreasureMap.askOldScholar(gold))   // 降级
    .then((s) => retry(() => TreasureMap.findTempleKey(s), 3))
    .then((k) => TreasureMap.readStarChart(k))
    .then((m) => TreasureMap.searchTemple(m.text))
    .then((f) => f.guard
      ? TreasureMap.fightGuard(hero)                 // 战斗分支
      : f)
    .then(() => TreasureMap.openTreasureBox())
    .catch((e) => Bus.emit(EVT.LOG, { text: e.message }));
}
      /* bus.js：发布-订阅，让逻辑层与渲染层解耦 */
const Bus = (function () {
  const target = new EventTarget();       // 借用原生事件系统
  function on(type, handler) {
    const w = (e) => handler(e.detail);
    target.addEventListener(type, w);
    return () => target.removeEventListener(type, w);
  }
  function emit(type, detail) {
    target.dispatchEvent(new CustomEvent(type, { detail }));
  }
  return { on, emit };
})();

/* 逻辑层：只广播，不知道谁在听 */
Bus.emit("item:found", clue);

/* 显示层：订阅自己关心的事件 */
